namespace = painted_galaxy_fe

event = {
	id = painted_galaxy_fe.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_maybe_painted_galaxy = yes
	}
	immediate = {
		# set FE type to random if missing required_dlc
		if = {
			limit = { has_biogenesis_dlc = no }
			every_system = {
				limit = { has_star_flag = painted_galaxy_fe_spawn_hive }
				remove_star_flag = painted_galaxy_fe_spawn_hive
				set_star_flag = painted_galaxy_fe_spawn_random
			}
		}

		# set defaults
		every_system = {
			limit = {
				has_star_flag = painted_galaxy_fe_spawn
				NOR = {
					has_star_flag = painted_galaxy_fe_spawn_random
					has_star_flag = painted_galaxy_fe_spawn_spiritualist
					has_star_flag = painted_galaxy_fe_spawn_materialist
					has_star_flag = painted_galaxy_fe_spawn_xenophile
					has_star_flag = painted_galaxy_fe_spawn_xenophobe
					has_star_flag = painted_galaxy_fe_spawn_machine
					has_star_flag = painted_galaxy_fe_spawn_hive
				}
			}
			set_star_flag = painted_galaxy_fe_spawn_random
		}
		every_system = {
			limit = {
				has_star_flag = painted_galaxy_fe_spawn
				NOR = {
					has_star_flag = painted_galaxy_fe_spawn_distance_30
					has_star_flag = painted_galaxy_fe_spawn_distance_40
					has_star_flag = painted_galaxy_fe_spawn_distance_50
					has_star_flag = painted_galaxy_fe_spawn_distance_60
					has_star_flag = painted_galaxy_fe_spawn_distance_70
					has_star_flag = painted_galaxy_fe_spawn_distance_80
					has_star_flag = painted_galaxy_fe_spawn_distance_90
					has_star_flag = painted_galaxy_fe_spawn_distance_100
					has_star_flag = painted_galaxy_fe_spawn_distance_110
					has_star_flag = painted_galaxy_fe_spawn_distance_120
					has_star_flag = painted_galaxy_fe_spawn_distance_130
					has_star_flag = painted_galaxy_fe_spawn_distance_140
					has_star_flag = painted_galaxy_fe_spawn_distance_150
					has_star_flag = painted_galaxy_fe_spawn_distance_160
					has_star_flag = painted_galaxy_fe_spawn_distance_170
					has_star_flag = painted_galaxy_fe_spawn_distance_180
					has_star_flag = painted_galaxy_fe_spawn_distance_190
					has_star_flag = painted_galaxy_fe_spawn_distance_200
				}
			}
			set_star_flag = painted_galaxy_fe_spawn_distance_40
		}

		set_spawn_system_batch = begin
		# global event country hasn't been created yet, just use a random one
		random_country = {
			export_trigger_value_to_variable = {
				trigger = count_system
				parameters = {
					limit = { painted_galaxy_is_fe_preferred_spawn = yes }
				}
				variable = painted_galaxy_available_fe_preferred_spawns
			}
			export_trigger_value_to_variable = {
				trigger = count_system
				parameters = {
					limit = { painted_galaxy_is_fe_highly_preferred_spawn = yes }
				}
				variable = painted_galaxy_available_fe_highly_preferred_spawns
			}
			set_variable = {
				which = painted_galaxy_fe_spawned
				value = 0
			}
			while = {
				limit = {
					check_galaxy_setup_value = {
						setting = num_fallen_empires
						value > painted_galaxy_fe_spawned
					}
				}
				add_variable = {
					which = painted_galaxy_fe_spawned
					value = 1
				}

				random_system = {
					limit = {
						has_star_flag = painted_galaxy_fe_spawn
						OR = {
							painted_galaxy_is_fe_preferred_spawn = yes
							check_variable_arithmetic = {
								which = prev.painted_galaxy_available_fe_preferred_spawns
								value = 0
							}
						}
						OR = {
							painted_galaxy_is_fe_highly_preferred_spawn = yes
							check_variable_arithmetic = {
								which = prev.painted_galaxy_available_fe_highly_preferred_spawns
								value = 0
							}
						}
					}
					save_event_target_as = painted_galaxy_fe_origin
					random_list = {
						0 = {
							modifier = {
								add = 100
								OR = {
									has_star_flag = painted_galaxy_fe_spawn_materialist
									has_star_flag = painted_galaxy_fe_spawn_random
								}
							}
							modifier = {
								factor = 99999
								any_country = { has_origin = origin_scion }
							}
							modifier = {
								factor = 0
								any_country = { has_country_flag = fallen_empire_1 }
							}
							painted_galaxy_fe_spawned_update_flags_and_variables = yes
							system_event = { id = painted_galaxy_fe.10 }
						}
						0 = {
							modifier = {
								add = 100
								OR = {
									has_star_flag = painted_galaxy_fe_spawn_spiritualist
									has_star_flag = painted_galaxy_fe_spawn_random
								}
							}
							modifier = {
								factor = 99999
								any_country = { has_origin = origin_scion }
							}
							modifier = {
								factor = 0
								any_country = { has_country_flag = fallen_empire_2 }
							}
							painted_galaxy_fe_spawned_update_flags_and_variables = yes
							system_event = { id = painted_galaxy_fe.20 }
						}
						0 = {
							modifier = {
								add = 100
								OR = {
									has_star_flag = painted_galaxy_fe_spawn_xenophile
									has_star_flag = painted_galaxy_fe_spawn_random
								}
							}
							modifier = {
								factor = 0
								any_country = { has_country_flag = fallen_empire_3 }
							}
							painted_galaxy_fe_spawned_update_flags_and_variables = yes
							system_event = { id = painted_galaxy_fe.30 }
						}
						0 = {
							modifier = {
								add = 100
								OR = {
									has_star_flag = painted_galaxy_fe_spawn_xenophobe
									has_star_flag = painted_galaxy_fe_spawn_random
								}
							}
							modifier = {
								factor = 0
								any_country = { has_country_flag = fallen_empire_4 }
							}
							painted_galaxy_fe_spawned_update_flags_and_variables = yes
							system_event = { id = painted_galaxy_fe.40 }
						}
						0 = {
							modifier = {
								add = 100
								OR = {
									has_star_flag = painted_galaxy_fe_spawn_machine
									has_star_flag = painted_galaxy_fe_spawn_random
								}
							}
							modifier = {
								factor = 0
								any_country = { has_country_flag = fallen_empire_machine }
							}
							painted_galaxy_fe_spawned_update_flags_and_variables = yes
							system_event = { id = painted_galaxy_fe.50 }
						}
						0 = {
							modifier = {
								add = 100
								OR = {
									has_star_flag = painted_galaxy_fe_spawn_hive
									has_star_flag = painted_galaxy_fe_spawn_random
								}
							}
							modifier = {
								factor = 0
								has_global_flag = hive_fe_spawned
							}
							modifier = {
								factor = 0
								has_biogenesis_dlc = no
							}
							modifier = {
								factor = 5
								any_country = {
									is_ai = no
									country_uses_bio_ships = yes
								}
							}
							painted_galaxy_fe_spawned_update_flags_and_variables = yes
							system_event = { id = painted_galaxy_fe.60 }
						}
					}
				}
			}

			every_system = {
				limit = {
					has_star_flag = painted_galaxy_fe_spawn
					has_star_flag = painted_galaxy_fe_spawn_fallback
				}
				save_event_target_as = painted_galaxy_fe_origin
				system_event = { id = painted_galaxy_fe.70 }
			}

			clear_variable = painted_galaxy_fe_spawned
			clear_variable = painted_galaxy_available_fe_preferred_spawns
			clear_variable = painted_galaxy_available_fe_highly_preferred_spawns
		}
		set_spawn_system_batch = end
	}
}
