namespace = painted_galaxy_guaranteed_colony

# spawn guaranteed colonies according to settings
event = {
	id = painted_galaxy_guaranteed_colony.1
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		is_maybe_painted_galaxy = yes
	}
	immediate = {
		# first, reset any guaranteed colonies to barren
		every_system = {
			limit = {
				OR = {
					has_star_flag = neighbor_t1_first_colony
					has_star_flag = neighbor_t2_second_colony
				}
			}
			random_system_planet = {
				limit = {
					has_planet_flag = prescripted_ideal
					# as a precaution, don't reset colonized planets
					is_colony = no
				}
				change_pc = pc_barren
				reroll_planet = yes
			}
		}

		# then for each valid country, find or create guaranteed colony and set correct planet class
		set_spawn_system_batch = begin
		every_country = {
			limit = {
				is_country_type = default
				# if their ideal planet class is not a normal planet type, they're probably not supposed to get habitable planets
				# eg life seeded, void dwellers, shattered ring, etc
				OR = {
					ideal_planet_class = pc_arid
					ideal_planet_class = pc_desert
					ideal_planet_class = pc_savannah
					ideal_planet_class = pc_continental
					ideal_planet_class = pc_ocean
					ideal_planet_class = pc_tropical
					ideal_planet_class = pc_arctic
					ideal_planet_class = pc_alpine
					ideal_planet_class = pc_tundra
					ideal_planet_class = pc_ark
				}
				# specifically forbidden origins
				NOT = {
					has_origin = origin_doomsday
					has_origin = origin_mindwardens
					has_origin = origin_slavers # MSI, has special logic
				}
			}
			save_event_target_as = painted_galaxy_current_country

			capital_scope.solar_system = {
				log = "capital system [This.GetName]"
				# find existing guaranteed colony systems
				closest_system = {
					max_steps = 2
					limit = { has_star_flag = neighbor_t1_first_colony }
					save_global_event_target_as = painted_galaxy_guaranteed_t1
					log = "found guaranteed_t1 [This.GetName]"
				}
				closest_system = {
					max_steps = 2
					limit = { has_star_flag = neighbor_t2_second_colony }
					save_global_event_target_as = painted_galaxy_guaranteed_t2
					log = "found guaranteed_t2 [This.GetName]"
				}

				# create systems if needed
				if = {
					limit = {
						num_guaranteed_colonies >= 1
						NOT = { exists = event_target:painted_galaxy_guaranteed_t1 }
					}
					if = {
						limit = { has_star_flag = sol }
						painted_galaxy_spawn_connected_system = { INITIALIZER = sol_neighbor_t1_first_colony }
					} else = {
						painted_galaxy_spawn_connected_system = { INITIALIZER = neighbor_t1_first_colony }
					}
					last_created_system = {
						save_global_event_target_as = painted_galaxy_guaranteed_t1
						log = "created guaranteed_t1 [This.GetName]"
					}
				}
				if = {
					limit = {
						num_guaranteed_colonies >= 2
						NOT = { exists = event_target:painted_galaxy_guaranteed_t2 }
					}
					random_neighbor_system = {
						limit = {
							OR = {
								# avoid spawning t2 directly connected to t1
								NOT = { is_same_value = event_target:painted_galaxy_guaranteed_t1 }
								# but allow it if no other neighbors
								prev = { count_neighbor_system = { count = 1 } }
							}
						}
						if = {
							limit = { prev = { has_star_flag = sol } }
							painted_galaxy_spawn_connected_system = { INITIALIZER = sol_neighbor_t2_second_colony }
						} else = {
							painted_galaxy_spawn_connected_system = { INITIALIZER = neighbor_t2_second_colony }
						}
					}
					last_created_system = {
						save_global_event_target_as = painted_galaxy_guaranteed_t2
						log = "created guaranteed_t2 [This.GetName]"
					}
				}

				# change planet classes
				# also set up colonies if country is lost colony parent
				if = {
					limit = { num_guaranteed_colonies >= 1 }
					event_target:painted_galaxy_guaranteed_t1 = {
						random_system_planet = {
							limit = { has_planet_flag = prescripted_ideal }
							change_pc = event_target:painted_galaxy_current_country
							reroll_planet = yes
							if = {
								limit = {
									event_target:painted_galaxy_current_country = { has_country_flag = lost_colony_parent }
								}
								set_planet_flag = lost_colony_planet
								create_colony = {
									owner = event_target:painted_galaxy_current_country
								}
								colony = {
									generate_lost_colony_parent_colony_1 = yes
								}
								solar_system = {
									create_starbase = {
										size = starbase_outpost
										owner = event_target:painted_galaxy_current_country
									}
								}
							}
						}
					}
				}
				if = {
					limit = { num_guaranteed_colonies >= 2 }
					event_target:painted_galaxy_guaranteed_t2 = {
						random_system_planet = {
							limit = { has_planet_flag = prescripted_ideal }
							change_pc = event_target:painted_galaxy_current_country
							reroll_planet = yes
							if = {
								limit = {
									event_target:painted_galaxy_current_country = { has_country_flag = lost_colony_parent }
								}
								set_planet_flag = lost_colony_planet
								create_colony = {
									owner = event_target:painted_galaxy_current_country
								}
								colony = {
									generate_lost_colony_parent_colony_2 = yes
								}
								solar_system = {
									create_starbase = {
										size = starbase_outpost
										owner = event_target:painted_galaxy_current_country
									}
								}
							}
						}
					}
				}

				# clear references so they're not reused by another country
				# note: this is why we're using global event_targets here; non-global can't be cleared
				if = {
					limit = { exists = event_target:painted_galaxy_guaranteed_t1 }
					clear_global_event_target = painted_galaxy_guaranteed_t1
				}
				if = {
					limit = { exists = event_target:painted_galaxy_guaranteed_t2 }
					clear_global_event_target = painted_galaxy_guaranteed_t2
				}
			}
		}
		set_spawn_system_batch = end
	}
}
